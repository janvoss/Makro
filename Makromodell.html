Makromodell

1 Konsumfunktion

1.1 Verfügbares Einkommen

\[Y_t = w_t N_t + D_t\]

\[Y_{t+1} = w_{t+1} N_{t+1} + D_{t+1} + (1 + r_t) S_t\]

\[D_{t+1} = Y_{t+1} + (1 - \delta) K_{t+1} - w_{t+1} N_{t+1} - (1 + r_t) B_t^I\]

\[B_t^I = K_{t+1} - (1 - \delta) K_t=I_t\] \[Y_t=C_t+G_t+I_t\]

1.2 Konsum in \(t\)

\[C_t(r, W) = \frac{W}{1 + \beta^{\frac{1}{\sigma}} (1+r)^{\frac{1-\sigma}{\sigma}}}\]

\[C_{t+1} = (1 + r_t) \left[ w_t N_t + D_t - C_t \right] + w_{t+1} N_{t+1} + D_{t+1}\]

2 Produktion

2.1 Produktionsfunktion

\[\begin{align*} Y_t &= A_t K_t^{\alpha} N_t^{1-\alpha}, \quad 0 < \alpha < 1 \\ Y_{t+1} &= A_{t+1} K_{t+1}^{\alpha} N_{t+1}^{1-\alpha} \end{align*}\]

2.2 Arbeitsnachfrage

\[(1 - \alpha) A_t \left(\frac{K_t}{N_t}\right)^{\alpha} = w_t\]

NoteUmstellen der Arbeitsnachfragefunktion

\[ \begin{aligned} (1 - \alpha) A_t \left(\frac{K_t}{N_t}\right)^{\alpha} &= w_t \\ \left(\frac{K_t}{N_t}\right)^{\alpha} &= \frac{w_t}{(1 - \alpha) A_t} \\ \frac{K_t}{N_t} &= \left(\frac{w_t}{(1 - \alpha) A_t}\right)^{\frac{1}{\alpha}} \\ N^d_t &= K_t \left(\frac{(1 - \alpha) A_t}{w_t}\right)^{\frac{1}{\alpha}} \end{aligned} \]

\[N^D_t = K_t \left(\frac{(1 - \alpha) A_t}{w_t}\right)^{\frac{1}{\alpha}}\]

\[N^D_{t+1} = K_{t+1} \left(\frac{(1 - \alpha) A_{t+1}}{w_{t+1}}\right)^{\frac{1}{\alpha}}\]

2.3 Investitionsnachfrage

\[\alpha A_{t+1} \left(\frac{N_{t+1}}{K_{t+1}}\right)^{1 - \alpha} = r_t +\delta\]

mit: \[K_{t+1} = I_t + (1 - \delta) K_t\]

\[ \alpha A_{t+1} \left(\frac{N_{t+1}}{I_t + (1 - \delta) K_t}\right)^{1 - \alpha} = r_t +\delta\]

NoteUmstellen der Investitionsfunktion

\[ \begin{aligned} \alpha A_{t+1} \left(\frac{N_{t+1}}{I_t + (1 - \delta) K_t}\right)^{1 - \alpha} &= r_t + \delta \\ \left(\frac{N_{t+1}}{I_t + (1 - \delta) K_t}\right)^{1 - \alpha} &= \frac{r_t + \delta}{\alpha A_{t+1}} \\ \frac{N_{t+1}}{I_t + (1 - \delta) K_t} &= \left(\frac{r_t + \delta}{\alpha A_{t+1}}\right)^{\frac{1}{1 - \alpha}} \\ I_t + (1 - \delta) K_t &= N_{t+1} \left(\frac{\alpha A_{t+1}}{r_t + \delta}\right)^{\frac{1}{1 - \alpha}} \\ I_t &= N_{t+1} \left(\frac{\alpha A_{t+1}}{r_t + \delta}\right)^{\frac{1}{1 - \alpha}} - (1 - \delta) K_t \end{aligned} \]

\[I_t = N_{t+1} \left(\frac{\alpha A_{t+1}}{r_t + \delta}\right)^{\frac{1}{1 - \alpha}} - (1 - \delta) K_t\]

2.4 Arbeitsangebot

\[\theta_t \frac{1}{1 - N_t} = w_t\]

\[N^S_t = 1 - \frac{\theta_t}{w_t}\]

\[N^S_{t+1} = 1 - \frac{\theta_{t+1}}{w_{t+1}}\] [Korrekturen oben noch einpflegen]::

solve_macro_model <- function(
  K_t = 10, delta = 0.1,
  theta_t = 0.5, theta_t1 = 0.5,
  G_t = 0.5, G_t1 = 0.5,
  sigma = 1.0, beta = 0.96,
  A_t = 10, A_t1 = 10,
  alpha = 0.33
) {
  # --- 1. Periode t: Arbeitsmarkt & Produktion (unabhängig von r_t) ---
  labor_eq_t <- function(w) {
    Nd <- K_t * ((1 - alpha) * A_t / w)^(1 / alpha)
    Ns <- 1 - theta_t / w
    return(Nd - Ns)
  }
  w_t <- uniroot(labor_eq_t, interval = c(theta_t + 1e-6, 100))$root
  N_t <- 1 - theta_t / w_t
  Y_t <- A_t * K_t^alpha * N_t^(1 - alpha)
  D_t <- Y_t - w_t * N_t

  # --- 2. Funktionen in Abhängigkeit vom Realzins r_t ---
  eval_given_r <- function(r) {
    k_n_ratio <- (alpha * A_t1 / (r + delta))^(1 / (1 - alpha))
    w_t1 <- (1 - alpha) * A_t1 * (k_n_ratio)^alpha
    N_t1 <- 1 - theta_t1 / w_t1
    
    if (is.na(N_t1) || N_t1 <= 0) return(list(excess_demand = NA))
    
    K_t1 <- k_n_ratio * N_t1
    I_t  <- K_t1 - (1 - delta) * K_t
    B_I  <- I_t
    Y_t1 <- A_t1 * K_t1^alpha * N_t1^(1 - alpha)
    D_t1 <- Y_t1 + (1 - delta) * K_t1 - w_t1 * N_t1 - (1 + r) * B_I
    
    W <- (w_t * N_t + D_t - G_t) + (w_t1 * N_t1 + D_t1 - G_t1) / (1 + r)
    
    C_t  <- W / (1 + beta^(1 / sigma) * (1 + r)^((1 - sigma) / sigma))
    S_t  <- w_t * N_t + D_t - G_t - C_t
    C_t1 <- (1 + r) * S_t + w_t1 * N_t1 + D_t1 - G_t1
    
    excess_demand <- C_t + I_t + G_t - Y_t
    
    list(
      r_t = r, w_t = w_t, N_t = N_t, Y_t = Y_t, D_t = D_t,
      C_t = C_t, I_t = I_t, S_t = S_t, G_t = G_t,
      w_t1 = w_t1, N_t1 = N_t1, K_t1 = K_t1, Y_t1 = Y_t1,
      D_t1 = D_t1, C_t1 = C_t1, G_t1 = G_t1, W = W,
      excess_demand = excess_demand,
      check_t1 = Y_t1 + (1 - delta) * K_t1 - (C_t1 + G_t1)
    )
  }

  # Analytische Obergrenze r_max für N_{t+1} > 0
  r_max <- alpha * A_t1 * ((1 - alpha) * A_t1 / theta_t1)^((1 - alpha) / alpha) - delta
  r_min <- max(-delta + 1e-5, -0.99)
  
  if (r_max <= r_min) {
    stop("Kein positives Arbeitsangebot in t+1 möglich (theta_{t+1} zu hoch oder A_{t+1} zu niedrig).")
  }

  # Gleichgewichtszins r_t suchen
  r_star <- uniroot(
    function(r) eval_given_r(r)$excess_demand,
    interval = c(r_min, r_max - 1e-6)
  )$root
  
  return(eval_given_r(r_star))
}

  K_t      = 10.0,
  delta    = 0.10,
  theta_t  = 0.50,
  theta_t1 = 0.50,
  G_t      = 0.50,   # Baseline Staatsausgaben in t
  G_t1     = 0.50,   # Staatsausgaben in t+1
  sigma    = 1.00,
  beta     = 0.96,
  A_t      = 10.0,
  A_t1     = 10.0,
  alpha    = 0.33
)

# Schock-Parameter (leitet sich direkt aus params ab, nur G_t wird geändert)
params_shock <- modifyList(params, list(G_t = 0.80))

# ==============================================================================
# 2. MODELL-SOLVER (ohne eigene Parameter-Defaults)
# ==============================================================================
solve_macro_model <- function(p) {
  with(p, {
    # --- Schritt A: Periode t (Arbeitsmarkt & Produktion) ---
    labor_eq_t <- function(w) {
      Nd <- K_t * ((1 - alpha) * A_t / w)^(1 / alpha)
      Ns <- 1 - theta_t / w
      return(Nd - Ns)
    }
    w_t <- uniroot(labor_eq_t, interval = c(theta_t + 1e-6, 1000))$root
    N_t <- 1 - theta_t / w_t
    Y_t <- A_t * K_t^alpha * N_t^(1 - alpha)
    D_t <- Y_t - w_t * N_t

    # --- Schritt B: Hilfsfunktion für beliebiges r und beliebiges Einkommen Y_hyp ---
    eval_given_r <- function(r, Y_hyp = Y_t) {
      k_n_ratio <- (alpha * A_t1 / (r + delta))^(1 / (1 - alpha))
      w_t1 <- (1 - alpha) * A_t1 * (k_n_ratio)^alpha
      N_t1 <- 1 - theta_t1 / w_t1
      if (is.na(N_t1) || N_t1 <= 0) return(list(excess_demand = NA))

      K_t1 <- k_n_ratio * N_t1
      I_t  <- K_t1 - (1 - delta) * K_t
      B_I  <- I_t
      Y_t1 <- A_t1 * K_t1^alpha * N_t1^(1 - alpha)
      D_t1 <- Y_t1 + (1 - delta) * K_t1 - w_t1 * N_t1 - (1 + r) * B_I

      W <- (Y_hyp - G_t) + (w_t1 * N_t1 + D_t1 - G_t1) / (1 + r)
      mpc <- 1 / (1 + beta^(1 / sigma) * (1 + r)^((1 - sigma) / sigma))
      C_t <- mpc * W

      Yd <- C_t + I_t + G_t
      excess_demand <- Yd - Y_hyp

      PV_t1 <- (w_t1 * N_t1 + D_t1 - G_t1) / (1 + r)
      Y_IS  <- (mpc * (PV_t1 - G_t) + I_t + G_t) / (1 - mpc)

      S_t  <- Y_hyp - G_t - C_t
      C_t1 <- (1 + r) * S_t + w_t1 * N_t1 + D_t1 - G_t1

      list(
        r_t = r, w_t = w_t, N_t = N_t, Y_t = Y_t, D_t = D_t,
        C_t = C_t, I_t = I_t, S_t = S_t, G_t = G_t, Yd = Yd, Y_IS = Y_IS,
        w_t1 = w_t1, N_t1 = N_t1, K_t1 = K_t1, Y_t1 = Y_t1,
        D_t1 = D_t1, C_t1 = C_t1, G_t1 = G_t1, W = W,
        excess_demand = excess_demand,
        check_t1 = Y_t1 + (1 - delta) * K_t1 - (C_t1 + G_t1)
      )
    }

    r_max <- alpha * A_t1 * ((1 - alpha) * A_t1 / theta_t1)^((1 - alpha) / alpha) - delta
    r_min <- max(-delta + 1e-5, -0.99)
    r_star <- uniroot(
      function(r) eval_given_r(r, Y_hyp = Y_t)$excess_demand,
      interval = c(r_min, r_max - 1e-6)
    )$root

    out <- eval_given_r(r_star, Y_hyp = Y_t)
    out$eval_given_r <- eval_given_r
    out$r_max <- r_max
    out$params <- p
    return(out)
  })
}
